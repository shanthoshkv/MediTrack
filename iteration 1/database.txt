-- ============================================================
-- MediTrack HMS — Hospital RFID Management System
-- Database Schema v3.0
-- ============================================================

CREATE DATABASE IF NOT EXISTS rfid_inventory CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE rfid_inventory;

CREATE TABLE IF NOT EXISTS system_logs (
    id         INT AUTO_INCREMENT PRIMARY KEY,
    log_level  ENUM('info','warning','error','critical') NOT NULL DEFAULT 'info',
    source     VARCHAR(100) NOT NULL,
    message    TEXT NOT NULL,
    ip_address VARCHAR(45) NULL,
    user_id    INT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_level(log_level), INDEX idx_source(source), INDEX idx_created(created_at)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS admins (
    id         INT AUTO_INCREMENT PRIMARY KEY,
    full_name  VARCHAR(120) NOT NULL,
    username   VARCHAR(50) UNIQUE NOT NULL,
    password   VARCHAR(255) NOT NULL,
    email      VARCHAR(150) NULL,
    role       ENUM('super_admin','manager','viewer','biomedical','pharmacist','nurse_admin') DEFAULT 'viewer',
    department VARCHAR(100) NULL,
    status     ENUM('active','disabled') DEFAULT 'active',
    last_login DATETIME NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS suppliers (
    id             INT AUTO_INCREMENT PRIMARY KEY,
    name           VARCHAR(150) NOT NULL,
    contact_person VARCHAR(120) NULL,
    email          VARCHAR(150) NULL,
    phone          VARCHAR(30) NULL,
    address        TEXT NULL,
    city           VARCHAR(80) NULL,
    country        VARCHAR(80) DEFAULT 'India',
    gst_number     VARCHAR(20) NULL,
    supplier_type  ENUM('medical_equipment','pharmaceuticals','consumables','general','lab_reagents','it_equipment') DEFAULT 'general',
    rating         TINYINT DEFAULT 3,
    is_active      TINYINT(1) DEFAULT 1,
    notes          TEXT NULL,
    created_at     TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS locations (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    parent_id       INT NULL,
    location_code   VARCHAR(30) NULL UNIQUE,
    location_name   VARCHAR(120) NOT NULL,
    location_type   ENUM('building','floor','ward','room','bed','storage','pharmacy','lab','ot','icu','emergency','blood_bank','cssd','reception','corridor') DEFAULT 'ward',
    zone            VARCHAR(80) NOT NULL DEFAULT 'General',
    reader_id       VARCHAR(50) NULL UNIQUE,
    reader_type     ENUM('entry','exit','both') DEFAULT 'both',
    api_key         VARCHAR(100) NULL,
    capacity        INT DEFAULT 0,
    temperature_min DECIMAL(5,2) NULL,
    temperature_max DECIMAL(5,2) NULL,
    last_heartbeat  DATETIME NULL,
    is_active       TINYINT(1) DEFAULT 1,
    notes           TEXT NULL,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (parent_id) REFERENCES locations(id) ON DELETE SET NULL,
    INDEX idx_parent(parent_id), INDEX idx_reader(reader_id)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS categories (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    parent_id       INT NULL,
    name            VARCHAR(100) NOT NULL,
    code            VARCHAR(20) NULL UNIQUE,
    description     TEXT NULL,
    requires_expiry TINYINT(1) DEFAULT 0,
    FOREIGN KEY (parent_id) REFERENCES categories(id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS items (
    id                     INT AUTO_INCREMENT PRIMARY KEY,
    uid                    VARCHAR(50) UNIQUE NOT NULL COMMENT 'RFID Tag UID',
    item_name              VARCHAR(150) NOT NULL,
    sku                    VARCHAR(80) NULL UNIQUE,
    barcode                VARCHAR(80) NULL,
    serial_number          VARCHAR(100) NULL UNIQUE,
    item_type              VARCHAR(80) NOT NULL,
    item_category          VARCHAR(80) NOT NULL,
    category_id            INT NULL,
    brand                  VARCHAR(100) NULL,
    model_number           VARCHAR(100) NULL,
    description            TEXT NULL,
    quantity               INT NOT NULL DEFAULT 1,
    min_stock_level        INT DEFAULT 0,
    max_stock_level        INT DEFAULT 9999,
    reorder_quantity       INT DEFAULT 1,
    unit_of_measure        VARCHAR(30) DEFAULT 'unit',
    unit_cost              DECIMAL(12,2) DEFAULT 0.00,
    unit_price             DECIMAL(12,2) DEFAULT 0.00,
    currency               VARCHAR(5) DEFAULT 'INR',
    location_id            INT NULL,
    department             VARCHAR(120) NULL,
    assigned_to            VARCHAR(120) NULL,
    assigned_patient_id    INT NULL,
    bin_location           VARCHAR(50) NULL,
    supplier_id            INT NULL,
    purchase_order_no      VARCHAR(80) NULL,
    invoice_number         VARCHAR(80) NULL,
    purchase_date          DATE NULL,
    warranty_expiry        DATE NULL,
    manufacture_date       DATE NULL,
    expiry_date            DATE NULL,
    last_service_date      DATE NULL,
    next_service_date      DATE NULL,
    last_seen              DATETIME NULL,
    registered_date        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    lifecycle              ENUM('active','maintenance','expiring_soon','expired','retired','disposed') DEFAULT 'active',
    status                 ENUM('in_stock','checked_out','in_transit','quarantined','missing','damaged','under_repair','sterilizing') DEFAULT 'in_stock',
    condition_grade        ENUM('new','good','fair','poor','scrap') DEFAULT 'new',
    weight_kg              DECIMAL(10,3) NULL,
    dimensions_cm          VARCHAR(50) NULL,
    color                  VARCHAR(50) NULL,
    image_url              VARCHAR(255) NULL,
    is_serialized          TINYINT(1) DEFAULT 1,
    is_perishable          TINYINT(1) DEFAULT 0,
    requires_approval      TINYINT(1) DEFAULT 0,
    is_hazardous           TINYINT(1) DEFAULT 0,
    is_controlled_drug     TINYINT(1) DEFAULT 0,
    requires_sterilization TINYINT(1) DEFAULT 0,
    last_sterilized_at     DATETIME NULL,
    notes                  TEXT NULL,
    custom_fields          JSON NULL,
    FOREIGN KEY (location_id) REFERENCES locations(id) ON DELETE SET NULL,
    FOREIGN KEY (supplier_id) REFERENCES suppliers(id) ON DELETE SET NULL,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL,
    INDEX idx_uid(uid), INDEX idx_status(status), INDEX idx_lifecycle(lifecycle),
    INDEX idx_expiry(expiry_date), INDEX idx_location(location_id), INDEX idx_dept(department)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS patients (
    id                 INT AUTO_INCREMENT PRIMARY KEY,
    rfid_uid           VARCHAR(50) UNIQUE NULL,
    patient_id         VARCHAR(30) UNIQUE NOT NULL,
    full_name          VARCHAR(150) NOT NULL,
    dob                DATE NULL,
    age                INT NULL,
    gender             ENUM('male','female','other') NOT NULL DEFAULT 'male',
    blood_group        ENUM('A+','A-','B+','B-','AB+','AB-','O+','O-','Unknown') DEFAULT 'Unknown',
    phone              VARCHAR(20) NULL,
    emergency_contact  VARCHAR(120) NULL,
    emergency_phone    VARCHAR(20) NULL,
    address            TEXT NULL,
    admission_date     DATETIME NULL,
    discharge_date     DATETIME NULL,
    ward_id            INT NULL,
    bed_number         VARCHAR(20) NULL,
    attending_doctor   VARCHAR(120) NULL,
    diagnosis          TEXT NULL,
    admission_type     ENUM('emergency','elective','transfer','outpatient') DEFAULT 'elective',
    status             ENUM('admitted','discharged','transferred','icu','critical','deceased','outpatient') DEFAULT 'admitted',
    diet_type          VARCHAR(80) NULL,
    allergy_notes      TEXT NULL,
    insurance_provider VARCHAR(100) NULL,
    insurance_policy   VARCHAR(80) NULL,
    notes              TEXT NULL,
    created_by         VARCHAR(80) NULL,
    created_at         TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (ward_id) REFERENCES locations(id) ON DELETE SET NULL,
    INDEX idx_rfid(rfid_uid), INDEX idx_status(status), INDEX idx_ward(ward_id), INDEX idx_pid(patient_id)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS staff_members (
    id                 INT AUTO_INCREMENT PRIMARY KEY,
    rfid_uid           VARCHAR(50) UNIQUE NULL,
    employee_id        VARCHAR(30) UNIQUE NOT NULL,
    full_name          VARCHAR(150) NOT NULL,
    designation        VARCHAR(100) NULL,
    department         VARCHAR(120) NULL,
    phone              VARCHAR(20) NULL,
    email              VARCHAR(150) NULL,
    shift              ENUM('morning','afternoon','night','on_call') DEFAULT 'morning',
    status             ENUM('active','on_leave','inactive') DEFAULT 'active',
    last_seen_location INT NULL,
    last_seen_at       DATETIME NULL,
    notes              TEXT NULL,
    created_at         TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (last_seen_location) REFERENCES locations(id) ON DELETE SET NULL,
    INDEX idx_rfid(rfid_uid), INDEX idx_status(status)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS maintenance_logs (
    id               INT AUTO_INCREMENT PRIMARY KEY,
    item_id          INT NOT NULL,
    maintenance_type ENUM('preventive','corrective','calibration','inspection','sterilization','amc','installation') NOT NULL,
    scheduled_date   DATE NULL,
    performed_date   DATE NULL,
    performed_by     VARCHAR(120) NULL,
    vendor_name      VARCHAR(150) NULL,
    cost             DECIMAL(10,2) DEFAULT 0,
    status           ENUM('scheduled','in_progress','completed','overdue','cancelled') DEFAULT 'scheduled',
    findings         TEXT NULL,
    next_due_date    DATE NULL,
    certificate_no   VARCHAR(80) NULL,
    notes            TEXT NULL,
    created_by       VARCHAR(80) NULL,
    created_at       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (item_id) REFERENCES items(id) ON DELETE CASCADE,
    INDEX idx_item(item_id), INDEX idx_status(status), INDEX idx_scheduled(scheduled_date)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS blood_units (
    id               INT AUTO_INCREMENT PRIMARY KEY,
    unit_id          VARCHAR(30) UNIQUE NOT NULL,
    rfid_uid         VARCHAR(50) UNIQUE NULL,
    blood_group      ENUM('A+','A-','B+','B-','AB+','AB-','O+','O-') NOT NULL,
    component        ENUM('whole_blood','packed_rbc','plasma','platelets','cryoprecipitate','ffp') NOT NULL,
    volume_ml        INT NOT NULL DEFAULT 350,
    collection_date  DATE NOT NULL,
    expiry_date      DATE NOT NULL,
    donor_id         VARCHAR(30) NULL,
    status           ENUM('available','reserved','issued','discarded','quarantine','testing') DEFAULT 'available',
    location_id      INT NULL,
    issued_to_patient INT NULL,
    issued_at        DATETIME NULL,
    issued_by        VARCHAR(80) NULL,
    cross_match_done TINYINT(1) DEFAULT 0,
    notes            TEXT NULL,
    created_at       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (location_id) REFERENCES locations(id) ON DELETE SET NULL,
    FOREIGN KEY (issued_to_patient) REFERENCES patients(id) ON DELETE SET NULL,
    INDEX idx_group(blood_group), INDEX idx_status(status), INDEX idx_expiry(expiry_date)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS scan_logs (
    id               INT AUTO_INCREMENT PRIMARY KEY,
    uid              VARCHAR(50) NOT NULL,
    item_id          INT NULL,
    patient_id       INT NULL,
    staff_id         INT NULL,
    from_location_id INT NULL,
    to_location_id   INT NULL,
    reader_id        VARCHAR(50) NOT NULL,
    action_type      ENUM('scanned','transfer','checkout','return','manual_move','conflict_suppressed','cycle_count','disposal','entry','exit','staff_scan','patient_scan') NOT NULL,
    performed_by     VARCHAR(80) NULL,
    notes            TEXT NULL,
    scan_time        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_scan_time(scan_time), INDEX idx_uid_time(uid, scan_time),
    INDEX idx_item(item_id), INDEX idx_patient(patient_id), INDEX idx_action(action_type)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS approvals (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    item_id         INT NOT NULL,
    request_type    ENUM('checkout','checkin','transfer','retirement','disposal','repair','sterilization') NOT NULL,
    requested_by    VARCHAR(80) NOT NULL,
    target_assignee VARCHAR(80) NULL,
    from_location   INT NULL,
    to_location     INT NULL,
    reason          TEXT NULL,
    status          ENUM('pending','approved','rejected') DEFAULT 'pending',
    reviewed_by     VARCHAR(80) NULL,
    reviewed_at     DATETIME NULL,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (item_id) REFERENCES items(id),
    INDEX idx_status(status)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS physical_counts (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    count_code   VARCHAR(30) NULL UNIQUE,
    location_id  INT NOT NULL,
    initiated_by VARCHAR(80) NULL,
    expected_qty INT NOT NULL DEFAULT 0,
    scanned_qty  INT DEFAULT 0,
    variance     INT GENERATED ALWAYS AS (scanned_qty - expected_qty) STORED,
    status       ENUM('in_progress','completed','cancelled') DEFAULT 'in_progress',
    notes        TEXT NULL,
    started_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at DATETIME NULL,
    FOREIGN KEY (location_id) REFERENCES locations(id),
    INDEX idx_status(status)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS alerts (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    item_id      INT NULL,
    patient_id   INT NULL,
    alert_type   ENUM('unknown_tag','low_stock','expiry_warning','reader_offline','conflict','missing_asset','unauthorized_move','maintenance_due','patient_alert','blood_expiry','controlled_drug') DEFAULT 'unknown_tag',
    severity     ENUM('info','warning','critical') NOT NULL,
    message      VARCHAR(500) NOT NULL,
    action_taken VARCHAR(255) NULL,
    is_resolved  TINYINT(1) DEFAULT 0,
    resolved_by  VARCHAR(80) NULL,
    resolved_at  DATETIME NULL,
    created_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (item_id) REFERENCES items(id) ON DELETE SET NULL,
    INDEX idx_resolved(is_resolved), INDEX idx_severity(severity), INDEX idx_created(created_at)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS purchase_orders (
    id            INT AUTO_INCREMENT PRIMARY KEY,
    po_number     VARCHAR(50) UNIQUE NOT NULL,
    supplier_id   INT NULL,
    status        ENUM('draft','sent','confirmed','received','cancelled') DEFAULT 'draft',
    total_amount  DECIMAL(12,2) DEFAULT 0.00,
    currency      VARCHAR(5) DEFAULT 'INR',
    order_date    DATE NULL,
    expected_date DATE NULL,
    received_date DATE NULL,
    created_by    VARCHAR(80) NULL,
    notes         TEXT NULL,
    created_at    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (supplier_id) REFERENCES suppliers(id) ON DELETE SET NULL
) ENGINE=InnoDB;
